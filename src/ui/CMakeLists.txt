# Исполняемый файл приложения.

# В сборке со статическими плагинами исполняемый файл обязан сослаться на символ
# регистрации каждого плагина, иначе компоновщик выбросит его как неиспользуемый.
# Порождение файла оставляет plugins/*/CMakeLists.txt единственным местом объявления
# плагина: spotty_add_plugin() записывает имя класса, а этот код его считывает.
get_property(spotty_static_plugin_classes GLOBAL PROPERTY SPOTTY_STATIC_PLUGIN_CLASSES)
get_property(spotty_static_plugin_targets GLOBAL PROPERTY SPOTTY_STATIC_PLUGIN_TARGETS)

set(spotty_static_plugins_cpp "// Generated by src/ui/CMakeLists.txt. Do not edit.\n")
if(spotty_static_plugin_classes)
    string(APPEND spotty_static_plugins_cpp "#include <QtPlugin>\n\n")
    foreach(class_name IN LISTS spotty_static_plugin_classes)
        string(APPEND spotty_static_plugins_cpp "Q_IMPORT_PLUGIN(${class_name})\n")
    endforeach()
else()
    string(APPEND spotty_static_plugins_cpp
        "// This build loads plugins at runtime; nothing to import.\n")
endif()

set(spotty_static_plugins_file "${CMAKE_CURRENT_BINARY_DIR}/StaticPlugins.cpp")
file(GENERATE OUTPUT "${spotty_static_plugins_file}" CONTENT "${spotty_static_plugins_cpp}")

qt_add_executable(spotty
    WIN32 MACOSX_BUNDLE
    ../main.cpp
    "${spotty_static_plugins_file}"
    # Значок собран заранее скриптом tools/make_icons.sh из resources/icons/spotty-master.png;
    # регенерировать при каждой сборке незачем, дизайн иконки не меняется от сборки к сборке.
    $<$<PLATFORM_ID:Windows>:${CMAKE_SOURCE_DIR}/resources/icons/spotty.rc>
    AppContext.h
    Formatting.cpp
    Formatting.h
    InterfaceBar.cpp
    InterfaceBar.h
    InterfaceCombo.cpp
    InterfaceCombo.h
    InterfaceItemDelegate.cpp
    InterfaceItemDelegate.h
    InterfaceLabel.cpp
    InterfaceLabel.h
    MainWindow.cpp
    MainWindow.h
    SendBar.cpp
    SendBar.h
    TerminalView.cpp
    TerminalView.h
    dialogs/InterfaceSettingsDialog.cpp
    dialogs/InterfaceSettingsDialog.h
    dialogs/InterfaceSettingsPanel.cpp
    dialogs/InterfaceSettingsPanel.h
    dialogs/SettingsDialog.cpp
    dialogs/SettingsDialog.h
    panels/GeneratorPanel.cpp
    panels/GeneratorPanel.h
    panels/LoggingPanel.cpp
    panels/LoggingPanel.h
    panels/MacrosPanel.cpp
    panels/MacrosPanel.h
    panels/ShortcutDelegate.cpp
    panels/ShortcutDelegate.h
    panels/SearchPanel.cpp
    panels/SearchPanel.h
    theme/AnsiPalette.cpp
    theme/AnsiPalette.h
    theme/MdiCodepoints.h
    theme/MdiIcons.cpp
    theme/MdiIcons.h
    theme/PlatformChrome.h
    theme/PlatformChrome.cpp
    theme/ThemeManager.cpp
    theme/ThemeManager.h
)

# Оформление рамки окна средствами системы: цвет заголовка задаётся не через QSS, а
# системным вызовом, и на macOS для него нужен Objective-C. Реализация для Windows и для
# систем без такой возможности лежит в PlatformChrome.cpp, для macOS — здесь.
if(APPLE)
    target_sources(spotty PRIVATE theme/PlatformChrome_mac.mm)
    target_link_libraries(spotty PRIVATE "-framework AppKit")
endif()

# Пути ресурсов складываются из PREFIX и пути файла относительно BASE, поэтому шрифт
# доступен как ":/fonts/materialdesignicons-webfont.ttf", а тема — как
# ":/themes/spotty.qss". Отдельный .qrc не нужен: список файлов здесь и есть объявление.
qt_add_resources(spotty "spotty-resources"
    PREFIX "/"
    BASE "${CMAKE_SOURCE_DIR}/resources"
    FILES
        "${CMAKE_SOURCE_DIR}/resources/fonts/materialdesignicons-webfont.ttf"
        # Галочка флажка. Отдельным файлом, а не глифом из шрифта значков: подставить
        # изображение в ::indicator можно только через url(), а нарисовать глиф шрифтом
        # таблица стилей не умеет.
        "${CMAKE_SOURCE_DIR}/resources/icons/check.svg"
        # Стрелки списка и счётчика — по той же причине: подставить их в ::down-arrow
        # можно только через url().
        "${CMAKE_SOURCE_DIR}/resources/icons/chevron-down.svg"
        "${CMAKE_SOURCE_DIR}/resources/icons/chevron-up.svg"
        "${CMAKE_SOURCE_DIR}/resources/icons/chevron-down-muted.svg"
        "${CMAKE_SOURCE_DIR}/resources/icons/chevron-up-muted.svg"
        "${CMAKE_SOURCE_DIR}/resources/themes/spotty.qss"
)

target_include_directories(spotty PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

if(APPLE)
    # MACOSX_BUNDLE_ICON_FILE называет файл по имени, а не по пути; CMake находит его
    # среди источников цели благодаря MACOSX_PACKAGE_LOCATION и copy-параметрам ниже.
    target_sources(spotty PRIVATE "${CMAKE_SOURCE_DIR}/resources/icons/spotty.icns")
    set_source_files_properties(
        "${CMAKE_SOURCE_DIR}/resources/icons/spotty.icns"
        PROPERTIES MACOSX_PACKAGE_LOCATION "Resources"
    )
    set_target_properties(spotty PROPERTIES MACOSX_BUNDLE_ICON_FILE "spotty.icns")
endif()

# Переводы. Исходный язык строк — английский, поэтому файла spotty_en.ts нет: без
# перевода интерфейс уже английский.
#
# LinguistTools необязателен: без него собирается тот же комплект, только на английском.
# Отказываться от сборки из-за отсутствия переводчика было бы неоправданно.
if(Qt6LinguistTools_FOUND)
    qt_add_translations(spotty
        TS_FILES "${CMAKE_SOURCE_DIR}/resources/i18n/spotty_ru.ts"
        RESOURCE_PREFIX "/i18n"
        LUPDATE_OPTIONS -no-obsolete -locations none
    )
else()
    message(WARNING "Qt6 LinguistTools not found - Spotty will be English-only. "
                    "Install qttools to build translations.")
endif()

target_link_libraries(spotty PRIVATE
    Qt6::Widgets
    Spotty::Core
    ${spotty_static_plugin_targets}
)

set_target_properties(spotty PROPERTIES
    # Кладём результат в корень дерева сборки, а не в build/src/ui: исполняемый файл
    # должен оказаться там, где его ожидает найти SPOTTY_PLUGIN_OUTPUT_DIR относительно
    # каталога плагинов.
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    MACOSX_BUNDLE_BUNDLE_NAME "Spotty"
    MACOSX_BUNDLE_BUNDLE_VERSION "${SPOTTY_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.spotty.Spotty"
)

# Цель spotty-package: распространяемый комплект для текущей платформы.
include(SpottyPackaging)
spotty_setup_packaging(spotty)
